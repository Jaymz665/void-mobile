pre_configure() {
    local host_libexec="/usr/lib/qt6/libexec"
    local target_libexec="/usr/aarch64-linux-gnu/usr/lib64/qt6/libexec"
    mkdir -p "$target_libexec"

    local tools=(
        qmltyperegistrar
        qmlcachegen
        qmlplugindump
        qmlsc
        qmllint
    )

    for tool in "${tools[@]}"; do
        if [ -f "$host_libexec/$tool" ]; then
            echo "Copying $tool to target libexec..."
            cp "$host_libexec/$tool" "$target_libexec/$tool"
            chmod +x "$target_libexec/$tool"
        else
            echo "Warning: $tool not found in host libexec, skipping"
        fi
    done
}
